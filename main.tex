  \infer[Num]{ }{\Theta; \Gamma \types \num{n}:\intty} \and

  \infer[Dict]
  {(\forall i \in [1,n])\,\Theta; \Gamma \types e_i: \anyty \\  \Theta; \Gamma \types e_i':Any }
  {\Theta; \Gamma \types \dictExp{e_1: e_1', \dots, e_n:e_n' }:\dictty } \and

  \infer[List]
  {(\forall i \in [1,n])\,\Theta; \Gamma \types e_i: \anyty}
  {\Theta; \Gamma \types \listExp{e_1, \dots, e_n}:\listty } \and

  \infer[New]
  { }
  {\Theta; \Gamma \types \newExp : \objty{i} } \,\,
  \objty{i} \, \text{is fresh} \and

  \infer[Ind-1]
  {\Theta; \Gamma \types e_1: \dictty}
  {\Theta; \Gamma \types e_1[e_2]: \anyty} \and

  \infer[Ind-2]
  {\Theta; \Gamma \types e_1: \listty \\ \Theta; \Gamma \types e_2:\intty}
  {\Theta; \Gamma \types e_1[e_2]: \anyty} \and

  \infer[Attr]
  {\Theta, \objty{i}.attr : A; \Gamma \types e: \objty{i} }
  {\Theta, \objty{i}.attr : A; \Gamma \types e.attr : A}
\end{mathpar}
